cmake_minimum_required(VERSION 3.21)
project(hashmap_locality LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(HASHBENCH_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(CTest)

add_library(hashbench INTERFACE)
target_include_directories(hashbench INTERFACE "${PROJECT_SOURCE_DIR}/include")
target_include_directories(hashbench SYSTEM INTERFACE "${PROJECT_SOURCE_DIR}/third_party")
find_path(VALGRIND_INCLUDE_DIR valgrind/cachegrind.h)
if(VALGRIND_INCLUDE_DIR)
    target_include_directories(hashbench SYSTEM INTERFACE "${VALGRIND_INCLUDE_DIR}")
endif()
target_compile_options(hashbench INTERFACE -Wall -Wextra -Wpedantic -g)
if(HASHBENCH_SANITIZERS)
    target_compile_options(hashbench INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(hashbench INTERFACE -fsanitize=address,undefined)
endif()
add_executable(hashmap_bench src/benchmark.cpp)
target_link_libraries(hashmap_bench PRIVATE hashbench)

if(BUILD_TESTING)
    find_package(GTest CONFIG QUIET)
    if(NOT GTest_FOUND)
        include(FetchContent)
        FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git
            GIT_TAG 52eb8108c5bdec04579160ae17225d66034bd723)
        set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
        set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
        FetchContent_MakeAvailable(googletest)
    endif()
    add_executable(hashmap_tests tests/maps_test.cpp)
    target_link_libraries(hashmap_tests PRIVATE hashbench GTest::gtest_main)
    include(GoogleTest)
    gtest_discover_tests(hashmap_tests DISCOVERY_MODE PRE_TEST)
endif()
